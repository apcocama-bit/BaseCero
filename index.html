Código del Backend (Code.gs en Google Apps Script) Crea un script en tu Google Drive vinculado a vuestra hoja de cálculo y pega este código: // GOOGLE APPS SCRIPT PARA PRESUPUESTO FAMILIAR YNAB const PINS = { andy: "1234", mujer: "5678" }; function doPost(e) { try { const request = JSON.parse(e.postData.contents); const user = request.user; const pin = request.pin; const workspace = request.workspace; // 'comun', 'personal_andy', 'personal_mujer' // VALIDACIÓN DE SEGURIDAD if (!PINS[user] || PINS[user] !== pin) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "PIN no válido" })) .setMimeType(ContentService.MimeType.JSON); } // BLOQUEO PRIVADO if (user === 'andy' && workspace === 'personal_mujer') { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Acceso denegado a datos privados." })) .setMimeType(ContentService.MimeType.JSON); } if (user === 'mujer' && workspace === 'personal_andy') { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Acceso denegado a datos privados." })) .setMimeType(ContentService.MimeType.JSON); } const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(workspace.toUpperCase()); if (!sheet) { sheet = ss.insertSheet(workspace.toUpperCase()); sheet.appendRow(["Fecha", "Tipo", "Cuenta", "Sobre", "Tercero", "Importe"]); } if (request.action === "saveTransaction") { const tx = request.transaction; sheet.appendRow([tx.date, tx.type, tx.account, tx.envelope, tx.payee, tx.amount]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
